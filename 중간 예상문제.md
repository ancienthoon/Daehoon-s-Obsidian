```python
def f(n, step=2):
    r = []
    for i in range(1, n, step):
        if i % 3 == 0:
            continue
        r.append(i * 2)
    return r, sum(r) / 2

a, b = f(10)
print(a, b)
print(f(7, step=3))
```

```python
def calc(nums, step=1):
    total = 0
    out = []
    for i in range(0, len(nums), step):
        if nums[i] % 2 == 0:
            continue
        total += nums[i]
        out.append(nums[i] * 2)
    return total, out

data = [3, 4, 5, 6, 7, 8, 9]
a, b = calc(data)
c, d = calc(data, 2)
print(a, b)
print(c, d)
print(a / c, a // c, a % c)
print(calc(data[::-1], 3))
```

```python
import copy

class Item:
    count = 0
    def __init__(self, name, qty):
        self.name = name
        self.qty = qty
        Item.count += 1
    def __str__(self):
        return self.name + "x" + str(self.qty)
    def __repr__(self):
        return "I(" + self.name + ")"
    def __add__(self, other):
        return Item(self.name, self.qty + other.qty)
    def __eq__(self, other):
        return self.name == other.name and self.qty == other.qty

class Box:
    def __init__(self):
        self.items = []
    def put(self, item):
        self.items.append(item)
    def total(self):
        s = 0
        for it in self.items:
            s += it.qty
        return s

a = Item("pen", 2)
b = Item("pen", 3)
c = a + b
box1 = Box()
box1.put(a)
box1.put(c)
box2 = copy.deepcopy(box1)
box1.items[0].qty = 10
box1.put(b)
print(c, a)
print(box1.items)
print(box2.items, box2.items[0])
print(box1.total(), box2.total(), len(box1.items), len(box2.items))
print(c == Item("pen", 5), c is box1.items[1], Item.count)
```

```python
class Animal:
    count = 0
    def __init__(self, name):
        self.name = name
        Animal.count += 1
    def speak(self):
        print(self.name, "소리")
    def __repr__(self):
        return "A:" + self.name

class Dog(Animal):
    def __init__(self, name, age):
        super().__init__(name)
        self.age = age
    def speak(self):
        super().speak()
        print("멍멍", self.age + 1)

class Puppy(Dog):
    def speak(self):
        print(self.name, "낑낑")

class Person:
    def __init__(self, name, pet):
        self.name = name
        self.pet = pet
    def makeSpeak(self):
        self.pet.speak()

zoo = [Animal("a"), Dog("b", 3), Puppy("c", 1)]
for z in zoo:
    z.speak()
p = Person("홍", zoo[1])
zoo[1].name = "bb"
p.makeSpeak()
q = Person("김", zoo[2])
q.pet.age = 7
q.makeSpeak()
print(zoo)
print(isinstance(zoo[2], Dog), isinstance(zoo[0], Puppy), type(zoo[2]) == Dog)
print(Animal.count, zoo[2].age, p.pet is zoo[1])
```
